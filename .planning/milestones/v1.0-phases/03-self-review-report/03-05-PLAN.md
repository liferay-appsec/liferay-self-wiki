    function todayUTC() {
      const now = new Date();
      return `${now.getUTCFullYear()}-${String(now.getUTCMonth() + 1).padStart(2, '0')}-${String(now.getUTCDate()).padStart(2, '0')}`;

    function resolveOutPath(rawOut, defaultPath) {
      if (!rawOut) return defaultPath;
      const r = resolve(rawOut);
      const vaultPrefix = resolve(getVaultPath()) + sep;
      if (!r.startsWith(vaultPrefix)) {
        process.stderr.write(`warn: --out path is outside the vault: ${r}\n`);

    // by iterating each in-cycle month and concatenating + dedup'ing.
    function weeksInRange(startISO, endISO) {
      const months = monthsInRange(startISO, endISO);
      const seen = new Set();
      const weeks = [];
      for (const m of months) {
        for (const w of weeksInMonth(m)) {
          if (!seen.has(w)) { seen.add(w); weeks.push(w); }
        }
      }
      return weeks;
    }

    // List every in-cycle ISO date (YYYY-MM-DD) — bounded by start/end inclusive.
    function datesInRange(startISO, endISO) {
      const months = monthsInRange(startISO, endISO);
      const all = [];
      for (const m of months) {
        for (const d of datesInMonth(m)) {
          if (d >= startISO && d <= endISO) all.push(d);
        }
      }

        review: {
          lastReviewedAt: today,
          lastReviewedCycle: window.cycleName,
        },
      });
    }

    import { applyUserConfig, ensureVaultConfigured } from '../core/config.js';
    import { selfReviewOrchestrator } from '../core/reviews.js';

    export async function selfReviewCommand(opts = {}) {
      await applyUserConfig();
      ensureVaultConfigured();

      // Mutex (project convention: validate at the command).
      // --since, --cycle, --last-cycle are mutually exclusive among themselves.
      // --prior-review and --out are orthogonal — they may co-exist with any.
      const windowFlags = [opts.since, opts.cycle, opts.lastCycle].filter((v) => v !== undefined && v !== null && v !== false);
      if (windowFlags.length > 1) {
        process.stderr.write('error: --since, --cycle, and --last-cycle are mutually exclusive\n');
        process.exit(1);
      }

      // --since must be a YYYY-MM-DD string.
      if (opts.since && !/^\d{4}-\d{2}-\d{2}$/.test(opts.since)) {
        process.stderr.write(`error: invalid --since value: ${opts.since} (expected YYYY-MM-DD)\n`);
        process.exit(1);
      }

      try {
        await selfReviewOrchestrator(opts);
      } catch (err) {
        // resolveReviewWindow throws with `invalid --cycle value:` for malformed cycle names;
        // surface those as exit-1 usage errors rather than uncaught rejections.
        if (err && typeof err.message === 'string' && err.message.startsWith('invalid --cycle value')) {
          process.stderr.write(`error: ${err.message}\n`);
          process.exit(1);
        }
        throw err;
      }
    }
